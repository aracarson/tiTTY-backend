Setup and Automation ScriptSave this script as setup_caddy_waf.sh, make it executable (chmod +x setup_caddy_waf.sh), and run it with sudo.bash#!/bin/bash
# setup_caddy_waf.sh
# Automates compiling Caddy + Coraza WAF and setting up API method whitelists.

set -e

# --- CONFIGURATION VARIABLES ---
CADDY_USER="caddy"
CADDY_CONFIG_DIR="/etc/caddy"
CADDY_BIN_PATH="/usr/local/bin/caddy"
DOMAIN_NAME="example.com"             # Change this to your domain
RUST_BACKEND="localhost:8080"         # Your Rust backend address

# Ensure script is run as root
if [ "$EUID" -ne 0 ]; then
  echo "❌ Please run this script using sudo or as root."
  exit 1
fi

echo "🚀 Step 1: Installing dependencies and Go..."
apt-get update && apt-get install -y curl git build-essential

# Install Go if not present
if ! command -v go &> /dev/null; then
  GO_VERSION="1.22.2"
  curl -OL "https://go.dev{GO_VERSION}.linux-amd64.tar.gz"
  rm -rf /usr/local/go && tar -C /usr/local -xzf "go${GO_VERSION}.linux-amd64.tar.gz"
  rm "go${GO_VERSION}.linux-amd64.tar.gz"
  export PATH=$PATH:/usr/local/go
  ln -sf /usr/local/go/bin/go /usr/local/bin/go
fi

echo "🛠️ Step 2: Compiling Caddy with Coraza WAF plugin via xcaddy..."
# Install xcaddy into a temporary directory
export GOPATH=/tmp/go
go install ://github.com

# Build the custom Caddy binary
/tmp/go/bin/xcaddy build --with github.com/corazawaf/coraza-caddy/v2
mv ./caddy "$CADDY_BIN_PATH"
chmod +x "$CADDY_BIN_PATH"

echo "👤 Step 3: Setting up system user and directories..."
if ! id -u "$CADDY_USER" &>/dev/null; then
  useradd --system --gid www-data --create-home --home-dir /var/lib/caddy --shell /usr/sbin/nologin "$CADDY_USER"
fi

mkdir -p "$CADDY_CONFIG_DIR"

echo "📝 Step 4: Generating the Caddyfile with Whitelist Rules..."
cat << 'EOF' > "$CADDY_CONFIG_DIR/Caddyfile"
{
    # Order Coraza execution at the very front of the middleware chain
    order coraza_waf first
}

# Macro definition for variable domain mapping
DOMAIN_PLACEHOLDER {
    # Coraza WAF processing block
    coraza_waf {
        # Loads the built-in OWASP Core Rule Set directly without file parsing overhead
        load_owasp_crs
        
        directives `
            SecRuleEngine On
            SecRequestBodyAccess On
            SecResponseBodyAccess On

            # =========================================================
            # CUSTOM WHITELIST LOGIC FOR SPECIFIC API METHODS
            # =========================================================
            
            # Rule 10001: Whitelist specific endpoints for safe HTTP GET methods
            # Skip all subsequent request phase checks if matching this condition
            SecRule REQUEST_URI "@startsWith /api/v1/public/" \
                "id:10001,phase:1,pass,nolog,allow,msg:'Whitelisted public API GET path'"

            # Rule 10002: Whitelist specific GraphQL mutations or health checks by URI matching
            SecRule REQUEST_URI "@streq /api/graphql/health" \
                "id:10002,phase:1,pass,nolog,allow,msg:'Whitelisted health status payload'"
                
            # Rule 10003: Example of method restriction (Only allow POST to specific secure endpoint)
            # If URI is /api/v1/secure-pay and method is NOT POST, block it immediately
            SecRule REQUEST_URI "@streq /api/v1/secure-pay" \
                "id:10003,phase:1,chain,deny,status:405,log,msg:'Invalid HTTP method used for payment gateway'"
                SecRule REQUEST_METHOD "!@streq POST"
        `
    }

    # Proxy clean streams through to your local Rust service
    reverse_proxy BACKEND_PLACEHOLDER
}
EOF

# Swap script tokens with variable parameters
sed -i "s/DOMAIN_PLACEHOLDER/$DOMAIN_NAME/g" "$CADDY_CONFIG_DIR/Caddyfile"
sed -i "s/BACKEND_PLACEHOLDER/$RUST_BACKEND/g" "$CADDY_CONFIG_DIR/Caddyfile"

chown -R "$CADDY_USER:www-data" "$CADDY_CONFIG_DIR"

echo "⚙️ Step 5: Creating the Systemd Service for Caddy..."
cat << EOF > /etc/systemd/system/caddy.service
[Unit]
Description=Caddy Web Server with Coraza WAF
Documentation=https://caddyserver.com/docs/
After=network.target network-online.target
Requires=network-online.target

[Service]
Type=notify
User=$CADDY_USER
Group=www-data
ExecStart=$CADDY_BIN_PATH run --environ --config $CADDY_CONFIG_DIR/Caddyfile
ExecReload=$CADDY_BIN_PATH reload --config $CADDY_CONFIG_DIR/Caddyfile --force
TimeoutStopSec=5s
LimitNOFILE=1048576
LimitNPROC=512
PrivateTmp=true
ProtectSystem=full
AmbientCapabilities=CAP_NET_BIND_SERVICE

[Install]
WantedBy=multi-user.target
EOF

echo "🔄 Step 6: Reloading systemd and starting Caddy..."
systemctl daemon-reload
systemctl enable caddy
systemctl restart caddy

echo "✅ Success! Custom Caddy + Coraza binary has been built and deployed."
echo "🔗 reverse_proxy pointing to: $RUST_BACKEND"
echo "📂 Caddyfile is located at: $CADDY_CONFIG_DIR/Caddyfile"
systemctl status caddy --no-pager | head -n 15
